# ============================================================================
# CMakeLists.txt —— 职工管理系统
#
# 【CMake 是什么】
#   CMake 本身不编译代码，它是一个"构建脚本生成器"：
#     1. 它读取本文件（CMakeLists.txt）
#     2. 生成对应平台的构建文件（Mac/Linux 默认生成 Makefile，也可生成 Xcode 工程）
#     3. 再由 make / Xcode 去调用编译器真正编译
#
# 【标准使用流程（在项目根目录下执行）】
#     mkdir build        # 新建 build 目录（只第一次需要）
#     cd build
#     cmake ..           # 让 CMake 读取上一级的 CMakeLists.txt，生成 Makefile
#     make               # 编译，生成可执行文件 worker_system
#     ./worker_system    # 运行
#
#   为什么要在 build/ 里编译？—— 这叫"外部构建"（out-of-source build），
#   所有编译产生的中间文件都关在 build/ 里，不污染源码目录；
#   想重新来过，直接删掉整个 build/ 即可。
# ============================================================================


# ----------------------------------------------------------------------------
# 1. 最低版本要求
#    如果电脑上的 CMake 低于这个版本，会直接报错停止，避免生成出错的构建文件。
# ----------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.16)


# ----------------------------------------------------------------------------
# 2. 项目声明
#    project(项目名 语言)
#      CXX 表示 C++（C 语言写 C）。
#    声明后 CMake 会自动定义一些变量，例如：
#      PROJECT_NAME            -> 项目名
#      PROJECT_SOURCE_DIR      -> 项目根目录（本文件所在目录）
# ----------------------------------------------------------------------------
project(WorkerManagerSystem CXX)


# ----------------------------------------------------------------------------
# 3. 设置 C++ 标准（三行通常成对出现）
# ----------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 11)          # 使用 C++11 标准（需要更高可改 14/17/20）
set(CMAKE_CXX_STANDARD_REQUIRED ON) # 编译器必须支持该标准，不支持就报错，
                                    # 而不是悄悄降级到旧标准
set(CMAKE_CXX_EXTENSIONS OFF)       # 只用标准 C++，禁用编译器私有扩展语法，
                                    # 保证代码换个编译器也能编译


# ----------------------------------------------------------------------------
# 4. 生成可执行文件
#    add_executable(目标名 源文件1 源文件2 ...)
#      - "目标名" 就是最终生成的程序的名字（这里是 worker_system）
#      - 路径是相对于本 CMakeLists.txt 所在目录写的
#      - 只需要列 .cpp 文件，.h 头文件不参与编译，不用列
#
#    注意：以后每新增一个 .cpp，都要加到这里，然后重新执行 cmake .. 和 make。
#
#    （进阶提示：网上常见 file(GLOB SRC "src/*.cpp") 自动收集源文件的写法，
#     简单但有大坑——新增文件后 CMake 感知不到，不会自动重新生成，
#     容易编译出旧版本的程序。学习阶段建议像下面这样手动列清楚。）
# ----------------------------------------------------------------------------
add_executable(worker_system
    src/职工管理系统.cpp
    src/employee.cpp
    src/workerManager.cpp
)


# ----------------------------------------------------------------------------
# 5. 头文件搜索路径  ★ 这就是解决"include 和 src 不在同一目录"的关键 ★
#
#    target_include_directories(目标名 PRIVATE 目录)
#      作用等价于 g++ 命令行里的 -I 参数：
#          g++ -Iinclude ...
#      有了它，编译 src/ 下的 .cpp 时，编译器就会去 include/ 目录里找头文件，
#      所以源文件里只要写：
#          #include "workerManager.h"     <- 只写文件名，这就是标准写法
#
#    三个关键字 PRIVATE / PUBLIC / INTERFACE 的区别（单可执行程序用 PRIVATE 即可）：
#      PRIVATE   只有这个目标自己用（可执行程序选它）
#      PUBLIC    自己用，链接本库的其他目标也用（写库时用）
#      INTERFACE 自己不编译，只给别人用（纯头文件库用）
#
#    注意：这条命令必须写在 add_executable 之后，
#    因为它是给"已经创建的目标"补充属性的。
#
#    （旧式写法 include_directories(include) 也能用，但它对之后定义的
#     所有目标全局生效，现代 CMake 更推荐上面这种按目标精确指定的方式。）
# ----------------------------------------------------------------------------
target_include_directories(worker_system PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)
# CMAKE_CURRENT_SOURCE_DIR 是 CMake 内置变量，表示"当前这个 CMakeLists.txt
# 所在的目录"（即项目根目录），拼接后得到 include 目录的绝对路径，
# 这样无论你在哪里执行 cmake，路径都不会错。
